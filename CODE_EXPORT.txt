## 2. Nâng cấp cốt lõi trong phiên bản M3.1

### 2.1. Quản lý trạng thái vật phẩm trên thế giới (World Pickup Persistence)
- **Vật phẩm đặt sẵn (Pre-placed Pickups):**
  - Gán `pickup_id` cố định: `pickup_village_potion_1`, `pickup_village_sword_1`, `pickup_village_amulet_1`.
  - Khi nhặt, ID được ghi vào `collected_pickup_ids` trong bản lưu. Khi tải game, nếu ID đã có trong danh sách đã nhặt, node tự động `queue_free()`, **không xuất hiện lần hai**.
- **Vật phẩm quái rơi (Dynamic Drops):**
  - Quái vật rơi đồ được gán cờ `is_dynamic = true` và `pickup_id` duy nhất dạng `drop_<item_id>_<timestamp>_<rand>`.
  - Nếu quái chết và rơi đồ nhưng Player chưa nhặt trước khi Lưu: Vị trí và số lượng vật phẩm rơi được lưu vào mảng `dropped_pickups`. Khi Tải, thế giới tái tạo chính xác các vật phẩm này trên mặt đất.
  - Khi nhặt đồ rơi thành công rồi Lưu/Tải: Vật phẩm KHÔNG bị nhân bản trên mặt đất.

### 2.2. Kiến trúc SaveManager nguyên tử (Atomic & Safe Save System)
- **Không bao giờ xóa file save cũ trước khi hoàn tất ghi:**
  1. Dữ liệu được ghi ra file tạm `.tmp`.
  2. Mở lại file tạm đọc và parse JSON để xác minh toàn vẹn cú pháp.
  3. Tạo bản sao lưu dự phòng `.bak` từ file save hiện tại (nếu có).
  4. Đổi tên file tạm `.tmp` đè lên file save chính thức.
  5. Nếu bất kỳ bước nào thất bại, hệ thống khôi phục ngay từ `.bak`, đảm bảo người chơi **không bao giờ bị mất save** khi xảy ra lỗi đột ngột.
- **Thẩm định dữ liệu sâu (Deep Validation & Clamping):**
  - Tọa độ Player: Giới hạn trong kích thước bản đồ hợp lệ `(-1000..5000, -500..1500)`.
  - HP / MP: Clamp trong khoảng `[0, Max]` (không bị âm hoặc vượt trần).
  - Vàng (Gold): Clamp `>= 0`.
  - Túi đồ: Bỏ qua slot rỗng, kiểm tra `ItemDatabase.has_item(id)` và clamp số lượng `[1, max_stack]`.
  - Trang bị: Kiểm tra bắt buộc đúng phân loại item (`WEAPON` cho slot vũ khí, `ACCESSORY` cho slot bùa/phụ kiện). Nếu file JSON bị sửa đổi hoặc gắn sai loại, tự động loại bỏ để chống crash.
  - Tiến độ nhiệm vụ: Validate `kill_count` trong khoảng `[0, required_kills]`.

### 2.3. Định danh kẻ địch ổn định (Stable Enemy ID Tracking)
- Kẻ địch trong thế giới có định danh chuỗi ổn định (`enemy_forest_1`, `enemy_forest_2`, `enemy_forest_3`).
- `QuestManager` lưu danh sách tiêu diệt bằng mảng chuỗi `_killed_enemy_ids: Array[String]`, không dựa vào `get_instance_id()` tạm thời của engine. Nhờ đó, trạng thái nhiệm vụ và danh sách quái đã diệt được đồng bộ hoàn hảo sau khi Save/Load.

### 2.4. Cách ly hoàn toàn dữ liệu kiểm thử (Test Isolation)
- Hỗ trợ `SaveManager.custom_save_path`: Toàn bộ các script test (`test_m3_1_flow.gd`, `test_m3_flow.gd`) sử dụng file test riêng (`user://noctelyra_m3_1_test.json`), tự động dọn dẹp sau khi chạy.
- File save thật của người chơi (`user://noctelyra_slot1.json`) được bảo toàn 100%, không bị ảnh hưởng hay ghi đè bởi bộ kiểm thử tự động.

## 4. Chi tiết 4 vật phẩm ban đầu

1. **Tinh thể Ma lực (`mana_shard`)**:
   - Loại: *Nguyên liệu (Material)*. Xếp chồng tối đa 99. Giá trị: 5 Vàng.
   - Rơi 100% từ sinh vật ma nhiễm trong Rừng Hắc Nguyệt.
2. **Bình Sinh Lực (`health_potion`)**:
   - Loại: *Tiêu hao (Consumable)*. Xếp chồng tối đa 20. Giá trị: 15 Vàng.
   - Hồi ngay lập tức 30 HP (giới hạn theo Max HP, không tràn máu). Giảm 1 bình sau khi dùng. Không thể uống khi Player đã chết (HP = 0).
3. **Kiếm Ánh Trăng (`moonsteel_sword`)**:
   - Loại: *Vũ khí (Weapon)*. Không xếp chồng (max_stack = 1). Giá trị: 100 Vàng.
   - Khi trang bị: Sát thương đánh thường tăng từ **25 lên 35**. Tháo ra hoàn trả chính xác về **25**.
4. **Bùa Nguyệt Quang (`moon_amulet`)**:
   - Loại: *Phụ kiện (Accessory)*. Không xếp chồng (max_stack = 1). Giá trị: 120 Vàng.
   - Khi trang bị: Max MP tăng từ **60 lên 80**. Tháo ra hoàn trả chính xác về **60** và tự động giới hạn Current MP về 60 (không tràn).

class_name Enemy
extends CharacterBody2D

signal state_changed(new_state_name: String)
signal enemy_defeated(enemy_node: Node)

enum State {
	IDLE,
	CHASE,
	ATTACK,
	HURT,
	DEAD
}

@export var move_speed: float = 120.0
@export var detection_range: float = 380.0
@export var attack_range: float = 55.0
@export var attack_cooldown: float = 1.2
@export var gravity: float = 1100.0
@export var enemy_id: String = ""
@export var loot_table: LootTable = null

@onready var health_component: HealthComponent = $HealthComponent
@onready var hurtbox: Hurtbox = $Hurtbox
@onready var attack_hitbox: Hitbox = $AttackHitbox
@onready var visual_root: Node2D = $VisualRoot
@onready var body_rect: ColorRect = $VisualRoot/BodyRect
@onready var eye_rect: ColorRect = $VisualRoot/EyeRect
@onready var hp_bar: ProgressBar = $VisualRoot/HPBar

var current_state: State = State.IDLE
var target_player: Player = null
var facing_direction: int = -1
var attack_timer: float = 0.0
var state_timer: float = 0.0
var _has_reported_death: bool = false

func _ready() -> void:
	if attack_hitbox:
		attack_hitbox.monitoring = false
		attack_hitbox.visible = false

	if health_component:
		health_component.health_changed.connect(_on_health_changed)
		health_component.died.connect(_on_died)
		if hp_bar:
			hp_bar.max_value = health_component.max_health
			hp_bar.value = health_component.current_health

	if hurtbox:
		hurtbox.hit_received.connect(_on_hit_received)

	_change_state(State.IDLE)

func _physics_process(delta: float) -> void:
	# Nếu đã DEAD thì không bao giờ chạy bất kỳ logic AI hay rơi nào khác
	if current_state == State.DEAD:
		velocity.x = move_toward(velocity.x, 0.0, 800.0 * delta)
		state_timer -= delta
		if visual_root:
			visual_root.modulate.a = max(0.0, state_timer / 1.0)
		if state_timer <= 0.0:
			queue_free()
		move_and_slide()
		return

	if attack_timer > 0.0:
		attack_timer -= delta

	if not is_on_floor():
		velocity.y += gravity * delta

	# Kiểm tra mục tiêu Player còn sống hay không
	if target_player and is_instance_valid(target_player):
		var p_health: HealthComponent = target_player.get_node_or_null("HealthComponent")
		if p_health and not p_health.is_alive():
			target_player = null # Bỏ mục tiêu nếu player đã chết
			if current_state == State.CHASE or current_state == State.ATTACK:
				_change_state(State.IDLE)

	match current_state:
		State.IDLE:
			velocity.x = move_toward(velocity.x, 0.0, 800.0 * delta)
			_find_target()
			if target_player and _is_target_in_range(detection_range):
				_change_state(State.CHASE)

		State.CHASE:
			if target_player == null or not is_instance_valid(target_player):
				_change_state(State.IDLE)
				return

			var dist := global_position.distance_to(target_player.global_position)
			if dist > detection_range * 1.4:
				_change_state(State.IDLE)
			elif dist <= attack_range:
				if attack_timer <= 0.0:
					_change_state(State.ATTACK)
				else:
					velocity.x = move_toward(velocity.x, 0.0, 800.0 * delta)
			else:
				var dir: float = sign(target_player.global_position.x - global_position.x)
				if dir != 0.0:
					_set_facing(int(dir))
					velocity.x = dir * move_speed

		State.ATTACK:
			velocity.x = move_toward(velocity.x, 0.0, 1000.0 * delta)
			state_timer -= delta
			if state_timer <= 0.0:
				_finish_attack()

		State.HURT:
			state_timer -= delta
			velocity.x = move_toward(velocity.x, 0.0, 900.0 * delta)
			if state_timer <= 0.0:
				if target_player:
					_change_state(State.CHASE)
				else:
					_change_state(State.IDLE)

	move_and_slide()

func _change_state(new_state: State) -> void:
	# Bảo đảm: Một khi đã DEAD thì KHÔNG THỂ bị ghi đè bởi bất kỳ state nào khác
	if current_state == State.DEAD:
		return

	current_state = new_state
	var sname := _get_state_name(current_state)
	emit_signal("state_changed", sname)

	match current_state:
		State.IDLE:
			if body_rect:
				body_rect.color = Color(0.8, 0.2, 0.2)
		State.CHASE:
			if body_rect:
				body_rect.color = Color(0.9, 0.35, 0.1)
		State.ATTACK:
			state_timer = 0.35
			attack_timer = attack_cooldown
			if body_rect:
				body_rect.color = Color(1.0, 0.8, 0.1)
			if attack_hitbox:
				attack_hitbox.reset_hits()
				attack_hitbox.monitoring = true
				attack_hitbox.visible = true
				attack_hitbox.position.x = abs(attack_hitbox.position.x) * float(facing_direction)
		State.HURT:
			state_timer = 0.25
			if body_rect:
				body_rect.color = Color(1.0, 1.0, 1.0)
			if attack_hitbox:
				attack_hitbox.monitoring = false
				attack_hitbox.visible = false
		State.DEAD:
			state_timer = 1.0
			if attack_hitbox:
				attack_hitbox.monitoring = false
				attack_hitbox.visible = false
			if hurtbox:
				hurtbox.set_deferred("monitoring", false)
				hurtbox.set_deferred("monitorable", false)
			collision_layer = 0
			_notify_kill_once()

func _notify_kill_once() -> void:
	if not _has_reported_death:
		_has_reported_death = true
		emit_signal("enemy_defeated", self)
		_spawn_loot()

func _spawn_loot() -> void:
	var spawn_parent = get_parent()
	if spawn_parent:
		if loot_table:
			loot_table.spawn_loot(global_position, spawn_parent)
		else:
			var pickup_scene: PackedScene = preload("res://src/loot/item_pickup.tscn")
			var pickup := pickup_scene.instantiate() as ItemPickup
			pickup.item_id = "mana_shard"
			pickup.amount = 1
			pickup.is_dynamic = true
			pickup.pickup_id = "drop_en_" + str(Time.get_ticks_usec()) + "_" + str(randi() % 100000)
			pickup.global_position = global_position
			spawn_parent.add_child.call_deferred(pickup)

func _finish_attack() -> void:
	if attack_hitbox:
		attack_hitbox.monitoring = false
		attack_hitbox.visible = false
	if target_player:
		_change_state(State.CHASE)
	else:
		_change_state(State.IDLE)

func _find_target() -> void:
	if target_player and is_instance_valid(target_player):
		return
	var players := get_tree().get_nodes_in_group("players")
	for p in players:
		var p_health: HealthComponent = p.get_node_or_null("HealthComponent")
		if p_health and p_health.is_alive():
			target_player = p as Player
			break

func _is_target_in_range(range_val: float) -> bool:
	if target_player and is_instance_valid(target_player):
		var p_health: HealthComponent = target_player.get_node_or_null("HealthComponent")
		if p_health and p_health.is_alive():
			return global_position.distance_to(target_player.global_position) <= range_val
	return false

func _set_facing(dir: int) -> void:
	if dir == 0 or dir == facing_direction:
		return
	facing_direction = dir
	if visual_root:
		visual_root.scale.x = float(-facing_direction)

func _on_health_changed(cur: float, max_v: float) -> void:
	if hp_bar:
		hp_bar.value = cur

func _on_hit_received(dmg: DamageInfo) -> void:
	if current_state == State.DEAD:
		return
	velocity.x = dmg.knockback_dir.x * dmg.knockback_force
	velocity.y = -120.0
	_change_state(State.HURT)

func _on_died() -> void:
	_change_state(State.DEAD)

func _get_state_name(st: State) -> String:
	match st:
		State.IDLE: return "IDLE"
		State.CHASE: return "CHASE"
		State.ATTACK: return "ATTACK"
		State.HURT: return "HURT"
		State.DEAD: return "DEAD"
		_: return "UNKNOWN"

class_name Player
extends CharacterBody2D

signal state_changed(new_state_name: String)
signal player_died()
signal player_respawned()

enum State {
	IDLE,
	RUN,
	JUMP,
	FALL,
	DASH,
	DEAD
}

@export var movement_data: PlayerMovementData

@onready var sprite_root: Node2D = $SpriteRoot
@onready var anim_player: AnimationPlayer = $AnimationPlayer
@onready var collision_shape: CollisionShape2D = $CollisionShape2D
@onready var health_component: HealthComponent = $HealthComponent
@onready var combat_component: CombatComponent = $CombatComponent
@onready var hurtbox: Hurtbox = $Hurtbox
@onready var melee_hitbox: Hitbox = $MeleeHitbox
@onready var inventory_component: InventoryComponent = $InventoryComponent
@onready var equipment_component: EquipmentComponent = $EquipmentComponent

var current_state: State = State.IDLE
var facing_direction: int = 1
var jumps_left: int = 2
var dash_timer: float = 0.0
var dash_cooldown_timer: float = 0.0
var dash_direction: int = 1
var spawn_position: Vector2 = Vector2.ZERO

func _ready() -> void:
	spawn_position = global_position
	if movement_data == null:
		movement_data = PlayerMovementData.new()
	_change_state(State.IDLE)

	if health_component:
		health_component.invulnerability_started.connect(_on_invuln_started)
		health_component.invulnerability_ended.connect(_on_invuln_ended)
		health_component.died.connect(_on_player_died)

func _physics_process(delta: float) -> void:
	# Khi chết: Vô hiệu hóa toàn bộ di chuyển, rơi và input
	if current_state == State.DEAD:
		velocity.x = move_toward(velocity.x, 0.0, 800.0 * delta)
		if not is_on_floor():
			velocity.y += movement_data.gravity * delta
		move_and_slide()
		return

	# Luôn giảm thời gian hồi chiêu Dash độc lập
	if dash_cooldown_timer > 0.0:
		dash_cooldown_timer -= delta

	# Kiểm tra nếu đang mở hội thoại NPC, vừa đóng, hoặc đang mở túi đồ thì tạm ngừng nhận input
	var ui: GameUI = _get_game_ui()
	if ui and (ui.is_dialogue_open() or ui.is_dialogue_cooling_down() or ui.is_inventory_open()):
		velocity.x = move_toward(velocity.x, 0.0, movement_data.friction * delta)
		move_and_slide()
		return

	# Xử lý input tấn công & ma pháp (J & K)
	_handle_combat_inputs()

	match current_state:
		State.DASH:
			# Trong lúc DASH: Giữ nguyên vận tốc lướt và không bị ghi đè bởi state khác
			dash_timer -= delta
			velocity.x = dash_direction * movement_data.dash_speed
			velocity.y = 0.0
			if dash_timer <= 0.0:
				if is_on_floor():
					var dir := _get_input_direction()
					if dir != 0.0:
						_change_state(State.RUN)
					else:
						_change_state(State.IDLE)
				else:
					_change_state(State.FALL)

		State.JUMP:
			# Ưu tiên Dash trên không hoặc Nhảy đôi
			if _check_dash():
				pass
			elif _check_double_jump():
				pass
			else:
				_apply_gravity(delta)
				_process_air_movement(delta)
				if velocity.y >= 0.0:
					_change_state(State.FALL)
				elif is_on_floor():
					_restore_jumps()
					if abs(velocity.x) > 10.0:
						_change_state(State.RUN)
					else:
						_change_state(State.IDLE)

		State.FALL:
			# Có thể Dash hoặc Nhảy đôi giữa lúc rơi
			if _check_dash():
				pass
			elif _check_double_jump():
				pass
			else:
				_apply_gravity(delta)
				_process_air_movement(delta)
				if is_on_floor():
					_restore_jumps()
					var dir := _get_input_direction()
					if dir != 0.0 or abs(velocity.x) > 10.0:
						_change_state(State.RUN)
					else:
						_change_state(State.IDLE)

		State.IDLE:
			_restore_jumps()
			if _check_dash():
				pass
			elif _check_jump():
				pass
			elif not is_on_floor():
				_change_state(State.FALL)
			else:
				_process_ground_movement(delta)
				if abs(velocity.x) > 10.0:
					_change_state(State.RUN)

		State.RUN:
			_restore_jumps()
			if _check_dash():
				pass
			elif _check_jump():
				pass
			elif not is_on_floor():
				_change_state(State.FALL)
			else:
				_process_ground_movement(delta)
				if abs(velocity.x) <= 10.0 and _get_input_direction() == 0.0:
					_change_state(State.IDLE)

	move_and_slide()

	# Khôi phục số lần nhảy khi đang chạm đất
	if is_on_floor() and current_state != State.JUMP and current_state != State.DEAD:
		_restore_jumps()

func _handle_combat_inputs() -> void:
	if combat_component == null:
		return
	if Input.is_action_just_pressed("attack"):
		combat_component.perform_attack(facing_direction)
	elif Input.is_action_just_pressed("cast_spell"):
		combat_component.cast_spell(self, facing_direction, health_component)

func _on_invuln_started() -> void:
	if sprite_root and current_state != State.DEAD:
		sprite_root.modulate = Color(1.0, 0.4, 0.4, 0.7)

func _on_invuln_ended() -> void:
	if sprite_root and current_state != State.DEAD:
		sprite_root.modulate = Color(1.0, 1.0, 1.0, 1.0)

func _on_player_died() -> void:
	_change_state(State.DEAD)
	velocity = Vector2.ZERO
	# Tắt hitbox tấn công
	if melee_hitbox:
		melee_hitbox.monitoring = false
		melee_hitbox.visible = false
	if combat_component:
		combat_component.end_attack()
	# Tắt nhận đòn
	if hurtbox:
		hurtbox.set_deferred("monitoring", false)
		hurtbox.set_deferred("monitorable", false)
	# Hiệu ứng mờ dần
	if sprite_root:
		sprite_root.modulate = Color(0.6, 0.2, 0.2, 0.6)

	emit_signal("player_died")

	var ui := _get_game_ui()
	if ui:
		ui.show_game_over()

func respawn() -> void:
	global_position = spawn_position
	velocity = Vector2.ZERO
	if health_component:
		health_component.current_health = health_component.max_health
		health_component.current_mana = health_component.max_mana
		health_component.is_invulnerable = false
		health_component.emit_signal("health_changed", health_component.current_health, health_component.max_health)
		health_component.emit_signal("mana_changed", health_component.current_mana, health_component.max_mana)
	if hurtbox:
		hurtbox.set_deferred("monitoring", true)
		hurtbox.set_deferred("monitorable", true)
	if sprite_root:
		sprite_root.modulate = Color(1.0, 1.0, 1.0, 1.0)
	
	_restore_jumps()
	dash_timer = 0.0
	dash_cooldown_timer = 0.0
	_change_state(State.IDLE)
	emit_signal("player_respawned")

func _restore_jumps() -> void:
	jumps_left = movement_data.max_jumps

func _change_state(new_state: State) -> void:
	current_state = new_state
	var state_name := _get_state_name(current_state)
	emit_signal("state_changed", state_name)

	var has_player := (anim_player != null)

	match current_state:
		State.IDLE:
			if has_player and anim_player.has_animation("idle"):
				anim_player.play("idle")
		State.RUN:
			if has_player and anim_player.has_animation("run"):
				anim_player.play("run")
		State.JUMP:
			if has_player and anim_player.has_animation("jump"):
				anim_player.play("jump")
			else:
				_play_fallback_air_visual()
		State.FALL:
			if has_player and anim_player.has_animation("fall"):
				anim_player.play("fall")
			else:
				_play_fallback_air_visual()
		State.DASH:
			if has_player and anim_player.has_animation("dash"):
				anim_player.play("dash")
			else:
				_play_fallback_dash_visual()
			dash_timer = movement_data.dash_duration
			dash_cooldown_timer = movement_data.dash_cooldown
			
			var move_dir := int(sign(_get_input_direction()))
			if move_dir != 0:
				dash_direction = move_dir
				_set_facing(dash_direction)
			else:
				dash_direction = facing_direction
		State.DEAD:
			if anim_player:
				anim_player.pause()

func _play_fallback_air_visual() -> void:
	if anim_player and anim_player.has_animation("run"):
		anim_player.play("run")
		anim_player.seek(0.12, true)
		anim_player.pause()

func _play_fallback_dash_visual() -> void:
	if anim_player and anim_player.has_animation("run"):
		anim_player.play("run")
		anim_player.seek(0.48, true)
		anim_player.pause()

func _get_input_direction() -> float:
	return Input.get_axis("move_left", "move_right")

func _process_ground_movement(delta: float) -> void:
	var direction := _get_input_direction()
	if direction != 0.0:
		_set_facing(int(sign(direction)))
		velocity.x = move_toward(velocity.x, direction * movement_data.move_speed, movement_data.acceleration * delta)
	else:
		velocity.x = move_toward(velocity.x, 0.0, movement_data.friction * delta)

func _process_air_movement(delta: float) -> void:
	var direction := _get_input_direction()
	if direction != 0.0:
		_set_facing(int(sign(direction)))
		velocity.x = move_toward(velocity.x, direction * movement_data.move_speed, movement_data.air_acceleration * delta)
	else:
		velocity.x = move_toward(velocity.x, 0.0, movement_data.air_friction * delta)

func _apply_gravity(delta: float) -> void:
	velocity.y += movement_data.gravity * delta

func _check_jump() -> bool:
	if Input.is_action_just_pressed("jump"):
		if is_on_floor() or jumps_left == movement_data.max_jumps:
			velocity.y = movement_data.jump_velocity
			jumps_left = movement_data.max_jumps - 1
			_change_state(State.JUMP)
			return true
	return false

func _check_double_jump() -> bool:
	if Input.is_action_just_pressed("jump") and jumps_left > 0:
		velocity.y = movement_data.double_jump_velocity
		jumps_left -= 1
		_change_state(State.JUMP)
		return true
	return false

func _check_dash() -> bool:
	if Input.is_action_just_pressed("dash") and dash_cooldown_timer <= 0.0:
		_change_state(State.DASH)
		return true
	return false

func _set_facing(dir: int) -> void:
	if dir == 0 or dir == facing_direction:
		return
	facing_direction = dir
	if sprite_root:
		sprite_root.scale.x = float(facing_direction)

func _get_game_ui() -> GameUI:
	var scene = get_tree().current_scene
	if scene:
		return scene.get_node_or_null("GameUI") as GameUI
	return null

func _get_state_name(state: State) -> String:
	match state:
		State.IDLE: return "IDLE"
		State.RUN: return "RUN"
		State.JUMP: return "JUMP"
		State.FALL: return "FALL"
		State.DASH: return "DASH"
		State.DEAD: return "DEAD"
		_: return "UNKNOWN"

class_name CombatComponent
extends Node

signal attack_started()
signal attack_finished()
signal spell_cast()

@export var attack_cooldown: float = 0.35
@export var spell_cooldown: float = 0.6
@export var spell_mana_cost: float = 15.0
@export var melee_hitbox: Hitbox
@export var projectile_scene: PackedScene = preload("res://src/combat/magic_projectile.tscn")
@export var projectile_spawn_offset: Vector2 = Vector2(30, -40)

var _attack_timer: float = 0.0
var _spell_timer: float = 0.0
var _is_attacking: bool = false
var _attack_anim_duration: float = 0.25

func _ready() -> void:
	if melee_hitbox:
		melee_hitbox.monitoring = false
		melee_hitbox.visible = false

func _process(delta: float) -> void:
	if _attack_timer > 0.0:
		_attack_timer -= delta
	if _spell_timer > 0.0:
		_spell_timer -= delta

	if _is_attacking:
		_attack_anim_duration -= delta
		if _attack_anim_duration <= 0.0:
			end_attack()

func can_attack() -> bool:
	return _attack_timer <= 0.0 and not _is_attacking

func can_cast_spell(health_comp: HealthComponent) -> bool:
	if _spell_timer > 0.0 or _is_attacking:
		return false
	if health_comp and health_comp.current_mana < spell_mana_cost:
		return false
	return true

func perform_attack(facing_dir: int) -> bool:
	if not can_attack():
		return false

	_is_attacking = true
	_attack_timer = attack_cooldown
	_attack_anim_duration = 0.22

	if melee_hitbox:
		melee_hitbox.reset_hits()
		melee_hitbox.monitoring = true
		melee_hitbox.visible = true
		# Căn chỉnh hitbox theo hướng mặt
		melee_hitbox.position.x = abs(melee_hitbox.position.x) * float(facing_dir)

	emit_signal("attack_started")
	return true

func end_attack() -> void:
	_is_attacking = false
	if melee_hitbox:
		melee_hitbox.monitoring = false
		melee_hitbox.visible = false
	emit_signal("attack_finished")

func cast_spell(owner_node: Node2D, facing_dir: int, health_comp: HealthComponent) -> bool:
	if not can_cast_spell(health_comp):
		return false

	if health_comp:
		var success := health_comp.consume_mana(spell_mana_cost)
		if not success:
			return false

	_spell_timer = spell_cooldown

	if projectile_scene and owner_node:
		var proj := projectile_scene.instantiate() as MagicProjectile
		var spawn_pos := owner_node.global_position + Vector2(projectile_spawn_offset.x * float(facing_dir), projectile_spawn_offset.y)
		proj.global_position = spawn_pos
		proj.setup(Vector2(float(facing_dir), 0.0), owner_node)
		var spawn_root: Node = owner_node.get_parent()
		if owner_node.get_tree() and owner_node.get_tree().current_scene:
			spawn_root = owner_node.get_tree().current_scene
		if spawn_root:
			spawn_root.add_child.call_deferred(proj)

	emit_signal("spell_cast")
	return true

func is_attacking() -> bool:
	return _is_attacking

class_name HealthComponent
extends Node

signal health_changed(current: float, max_val: float)
signal mana_changed(current: float, max_val: float)
signal damaged(amount: float, source: Node)
signal died()
signal invulnerability_started()
signal invulnerability_ended()

@export var max_health: float = 100.0
@export var current_health: float = 100.0
@export var max_mana: float = 50.0
@export var current_mana: float = 50.0
@export var mana_regen_rate: float = 5.0 # MP hồi phục mỗi giây
@export var invulnerability_duration: float = 0.6 # Thời gian miễn nhiễm sát thương sau khi trúng đòn

var is_invulnerable: bool = false
var _invuln_timer: float = 0.0

func _ready() -> void:
	current_health = clamp(current_health, 0.0, max_health)
	current_mana = clamp(current_mana, 0.0, max_mana)
	emit_signal("health_changed", current_health, max_health)
	emit_signal("mana_changed", current_mana, max_mana)

func _process(delta: float) -> void:
	# Cập nhật thời gian miễn nhiễm sát thương
	if is_invulnerable:
		_invuln_timer -= delta
		if _invuln_timer <= 0.0:
			is_invulnerable = false
			emit_signal("invulnerability_ended")

	# Tự động hồi MP dần theo thời gian nếu chưa đầy
	if mana_regen_rate > 0.0 and current_mana < max_mana:
		current_mana = min(max_mana, current_mana + mana_regen_rate * delta)
		emit_signal("mana_changed", current_mana, max_mana)

func take_damage(damage: DamageInfo) -> bool:
	if is_invulnerable or current_health <= 0.0:
		return false

	var final_dmg := damage.amount
	current_health = max(0.0, current_health - final_dmg)
	emit_signal("health_changed", current_health, max_health)
	emit_signal("damaged", final_dmg, damage.source)

	if invulnerability_duration > 0.0:
		is_invulnerable = true
		_invuln_timer = invulnerability_duration
		emit_signal("invulnerability_started")

	if current_health <= 0.0:
		emit_signal("died")

	return true

func consume_mana(amount: float) -> bool:
	if current_mana >= amount:
		current_mana -= amount
		emit_signal("mana_changed", current_mana, max_mana)
		return true
	return false

func is_alive() -> bool:
	return current_health > 0.0

		if hurtbox.owner_node in _already_hit_nodes:

		var success := hurtbox.apply_damage(dmg_info)
		if success:
			_already_hit_nodes.append(hurtbox.owner_node)
			emit_signal("hit_landed", hurtbox)

class_name InteractionCoordinator

## Bộ điều phối input tập trung cho hệ thống tương tác và hội thoại (M2.1)

@export var game_ui: GameUI

func _ready() -> void:
	add_to_group("interaction_coordinator")
	_ensure_game_ui()

func _ensure_game_ui() -> void:
	if game_ui == null:
		var cur = get_tree().current_scene if get_tree() else null
		if cur:
			game_ui = cur.get_node_or_null("GameUI") as GameUI

func _unhandled_input(event: InputEvent) -> void:
	if not event.is_action_pressed("interact"):
		return

	_ensure_game_ui()

	# 1. Khi đang mở hội thoại: E chỉ chuyển sang câu tiếp theo hoặc đóng hội thoại
	if game_ui and game_ui.is_dialogue_open():
		game_ui.advance_dialogue()
		get_viewport().set_input_as_handled()
		return

	# 2. Khi hội thoại vừa đóng: grace cooldown ngăn chặn nhấn đúp re-open tức thì
	if game_ui and game_ui.is_dialogue_cooling_down():
		get_viewport().set_input_as_handled()
		return

	# 3. Khi chưa mở hội thoại: E dùng để kích hoạt tương tác với NPC gần nhất
	var target_npc := _find_nearby_interactable_npc()
	if target_npc:
		target_npc.trigger_interaction()
		get_viewport().set_input_as_handled()
		return

func _find_nearby_interactable_npc() -> NPC:
	var npcs = get_tree().get_nodes_in_group("npcs")
	for n in npcs:
		if n is NPC and n.is_interactable():
			return n
	return null

class_name EquipmentComponent
extends Node

## Quản lý trang bị của Player (Weapon & Accessory) cho NOCTELYRA (M3)
## Đảm bảo tính toán chỉ số dựa trên Base Stats + Modifiers, không sửa gốc vĩnh viễn.

signal equipment_changed(slot_type: String, item_id: String)

@export var player: Player

var base_attack_damage: float = 25.0
var base_max_mana: float = 60.0

var equipped_weapon: String = ""
var equipped_accessory: String = ""

func _ready() -> void:
	if player == null and get_parent() is Player:
		player = get_parent() as Player

	_record_base_stats()

func _record_base_stats() -> void:
	if player:
		if player.melee_hitbox:
			base_attack_damage = player.melee_hitbox.damage
		var p_health := player.get_node_or_null("HealthComponent") as HealthComponent
		if p_health:
			base_max_mana = p_health.max_mana

func get_equipped(slot_type: String) -> String:
	match slot_type:
		"weapon": return equipped_weapon
		"accessory": return equipped_accessory
		_: return ""

func is_slot_equipped(slot_type: String) -> bool:
	return not get_equipped(slot_type).is_empty()

func equip(item_id: String) -> String:
	if not ItemDatabase.has_item(item_id):
		return ""
	var def := ItemDatabase.get_item(item_id)
	var prev_item: String = ""

	match def.item_type:
		ItemDefinition.ItemType.WEAPON:
			prev_item = equipped_weapon
			equipped_weapon = item_id
			_recalculate_stats()
			emit_signal("equipment_changed", "weapon", item_id)
		ItemDefinition.ItemType.ACCESSORY:
			prev_item = equipped_accessory
			equipped_accessory = item_id
			_recalculate_stats()
			emit_signal("equipment_changed", "accessory", item_id)
		_:
			return "" # Không phải trang bị

	return prev_item

func unequip(slot_type: String) -> String:
	var old_item: String = ""
	match slot_type:
		"weapon":
			old_item = equipped_weapon
			equipped_weapon = ""
			_recalculate_stats()
			emit_signal("equipment_changed", "weapon", "")
		"accessory":
			old_item = equipped_accessory
			equipped_accessory = ""
			_recalculate_stats()
			emit_signal("equipment_changed", "accessory", "")
		_:
			return ""

	return old_item

func _recalculate_stats() -> void:
	var total_attack := base_attack_damage
	var total_max_mana := base_max_mana

	# Cộng chỉ số từ Weapon
	if not equipped_weapon.is_empty() and ItemDatabase.has_item(equipped_weapon):
		var w_def := ItemDatabase.get_item(equipped_weapon)
		total_attack += float(w_def.stat_modifiers.get("attack_bonus", 0.0))

	# Cộng chỉ số từ Accessory
	if not equipped_accessory.is_empty() and ItemDatabase.has_item(equipped_accessory):
		var a_def := ItemDatabase.get_item(equipped_accessory)
		total_max_mana += float(a_def.stat_modifiers.get("max_mana_bonus", 0.0))

	# Áp dụng chỉ số vào Player
	if player:
		if player.melee_hitbox:
			player.melee_hitbox.damage = total_attack

		var p_health := player.get_node_or_null("HealthComponent") as HealthComponent
		if p_health:
			p_health.max_mana = total_max_mana
			# Không tạo MP vượt giới hạn sau khi tháo trang bị
			if p_health.current_mana > total_max_mana:
				p_health.current_mana = total_max_mana
			p_health.emit_signal("mana_changed", p_health.current_mana, p_health.max_mana)

func to_dict() -> Dictionary:
	return {
		"weapon": equipped_weapon,
		"accessory": equipped_accessory
	}

func load_from_dict(data: Dictionary) -> void:
	equipped_weapon = str(data.get("weapon", ""))
	equipped_accessory = str(data.get("accessory", ""))

	# Kiểm tra tồn tại trong DB và đúng loại trang bị (M3.1 validation)
	if not ItemDatabase.has_item(equipped_weapon) or ItemDatabase.get_item(equipped_weapon).item_type != ItemDefinition.ItemType.WEAPON:
		equipped_weapon = ""
	if not ItemDatabase.has_item(equipped_accessory) or ItemDatabase.get_item(equipped_accessory).item_type != ItemDefinition.ItemType.ACCESSORY:
		equipped_accessory = ""

	_recalculate_stats()
	emit_signal("equipment_changed", "weapon", equipped_weapon)
	emit_signal("equipment_changed", "accessory", equipped_accessory)

class_name InventoryComponent
extends Node

## Quản lý dữ liệu túi đồ (Inventory) 16 ô cho NOCTELYRA (M3)
## Hoàn toàn tách biệt khỏi logic giao diện (UI)

signal inventory_updated()
signal item_added(item_id: String, amount: int)
signal item_removed(item_id: String, amount: int)
signal item_used(item_id: String)

@export var max_slots: int = 16

# Mảng chứa Dictionary các ô đồ: [{ "item_id": String, "amount": int }, ...]
var slots: Array[Dictionary] = []

func _ready() -> void:
	if slots.is_empty():
		_initialize_slots()

func _initialize_slots() -> void:
	slots.clear()
	for i in range(max_slots):
		slots.append({ "item_id": "", "amount": 0 })

func get_slot(index: int) -> Dictionary:
	if index >= 0 and index < slots.size():
		return slots[index]
	return { "item_id": "", "amount": 0 }

func is_slot_empty(index: int) -> bool:
	var s := get_slot(index)
	return s.item_id.is_empty() or s.amount <= 0

func can_add_item(item_id: String, amount: int = 1) -> bool:
	if amount <= 0 or not ItemDatabase.has_item(item_id):
		return false
	var item_def := ItemDatabase.get_item(item_id)
	var needed := amount

	# Kiểm tra các ô hiện có cùng loại nếu có thể stack
	if item_def.max_stack > 1:
		for s in slots:
			if s.item_id == item_id and s.amount < item_def.max_stack:
				var space: int = item_def.max_stack - s.amount
				needed -= space
				if needed <= 0:
					return true

	# Kiểm tra các ô trống còn lại
	for s in slots:
		if s.item_id.is_empty() or s.amount <= 0:
			needed -= item_def.max_stack
			if needed <= 0:
				return true

	return false

func add_item(item_id: String, amount: int = 1) -> int:
	if amount <= 0 or not ItemDatabase.has_item(item_id):
		return amount
	var item_def := ItemDatabase.get_item(item_id)
	var remaining := amount

	# 1. Điền vào các ô cùng loại chưa đầy nếu stack được
	if item_def.max_stack > 1:
		for i in range(slots.size()):
			if remaining <= 0:
				break
			var s := slots[i]
			if s.item_id == item_id and s.amount < item_def.max_stack:
				var can_take: int = min(remaining, item_def.max_stack - s.amount)
				s.amount += can_take
				remaining -= can_take

	# 2. Điền vào các ô trống mới
	for i in range(slots.size()):
		if remaining <= 0:
			break
		var s := slots[i]
		if s.item_id.is_empty() or s.amount <= 0:
			var can_take: int = min(remaining, item_def.max_stack)
			s.item_id = item_id
			s.amount = can_take
			remaining -= can_take

	var added_total: int = amount - remaining
	if added_total > 0:
		emit_signal("item_added", item_id, added_total)
		emit_signal("inventory_updated")

	return remaining

func remove_item_at(slot_idx: int, amount: int = 1) -> bool:
	if slot_idx < 0 or slot_idx >= slots.size() or amount <= 0:
		return false
	var s := slots[slot_idx]
	if s.item_id.is_empty() or s.amount <= 0:
		return false

	var item_id: String = s.item_id
	var to_remove: int = min(amount, s.amount)
	s.amount -= to_remove

	if s.amount <= 0:
		s.item_id = ""
		s.amount = 0

	emit_signal("item_removed", item_id, to_remove)
	emit_signal("inventory_updated")
	return true

func remove_item_by_id(item_id: String, amount: int = 1) -> bool:
	if amount <= 0 or not has_item(item_id, amount):
		return false
	var remaining := amount
	for i in range(slots.size() - 1, -1, -1):
		if remaining <= 0:
			break
		var s := slots[i]
		if s.item_id == item_id:
			var to_remove: int = min(remaining, s.amount)
			s.amount -= to_remove
			remaining -= to_remove
			if s.amount <= 0:
				s.item_id = ""
				s.amount = 0

	emit_signal("item_removed", item_id, amount)
	emit_signal("inventory_updated")
	return true

func has_item(item_id: String, amount: int = 1) -> bool:
	var count := get_item_count(item_id)
	return count >= amount

func get_item_count(item_id: String) -> int:
	var count: int = 0
	for s in slots:
		if s.item_id == item_id:
			count += s.amount
	return count

func use_item_at(slot_idx: int, player: Player) -> bool:
	if slot_idx < 0 or slot_idx >= slots.size() or player == null:
		return false
	var s := slots[slot_idx]
	if s.item_id.is_empty() or s.amount <= 0:
		return false

	var item_def := ItemDatabase.get_item(s.item_id)
	if item_def == null or item_def.item_type != ItemDefinition.ItemType.CONSUMABLE:
		return false

	# Kiểm tra trạng thái sống của Player
	var p_health := player.get_node_or_null("HealthComponent") as HealthComponent
	if p_health == null or not p_health.is_alive():
		return false # Không thể sử dụng bình khi đã chết

	# Áp dụng hiệu ứng
	var used := false
	if item_def.use_effect.has("heal_hp"):
		var heal_val: float = float(item_def.use_effect["heal_hp"])
		p_health.current_health = min(p_health.max_health, p_health.current_health + heal_val)
		p_health.emit_signal("health_changed", p_health.current_health, p_health.max_health)
		used = true

	if used:
		var item_id: String = s.item_id
		remove_item_at(slot_idx, 1)
		emit_signal("item_used", item_id)
		return true

	return false

func to_dict() -> Array:
	var arr: Array = []
	for s in slots:
		arr.append({
			"item_id": s.item_id,
			"amount": s.amount
		})
	return arr

func load_from_dict(arr: Array) -> void:
	_initialize_slots()
	for i in range(min(arr.size(), max_slots)):
		var entry = arr[i]
		if entry is Dictionary:
			var id: String = str(entry.get("item_id", ""))
			var amt: int = int(entry.get("amount", 0))
			if not id.is_empty() and amt > 0 and ItemDatabase.has_item(id):
				var def := ItemDatabase.get_item(id)
				slots[i].item_id = id
				slots[i].amount = clamp(amt, 1, def.max_stack)
			else:
				slots[i].item_id = ""
				slots[i].amount = 0
	emit_signal("inventory_updated")

					icon.texture = def.icon

	# 2. Cập nhật 2 ô trang bị

				accessory_slot_btn.icon = a_def.icon if a_def else null

	# Làm mới hiển thị chi tiết vật phẩm đang chọn
	if _selected_slot_idx >= 0:
		_select_slot(_selected_slot_idx)
	elif not _selected_equip_slot.is_empty():
		_select_equip_slot(_selected_equip_slot)

func _select_slot(index: int) -> void:
	_selected_slot_idx = index
	_selected_equip_slot = ""
	_ensure_player()

	if player == null or player.inventory_component == null:
		_clear_details()
		return

	var s := player.inventory_component.get_slot(index)
	if s.item_id.is_empty() or s.amount <= 0:
		_clear_details()
		return

	var def := ItemDatabase.get_item(s.item_id)
	if def == null:
		_clear_details()
		return

	_display_item_details(def, s.amount, false)

func _select_equip_slot(slot_type: String) -> void:
	_selected_slot_idx = -1
	_selected_equip_slot = slot_type
	_ensure_player()

	if player == null or player.equipment_component == null:
		_clear_details()
		return

	var item_id := player.equipment_component.get_equipped(slot_type)
	if item_id.is_empty():
		_clear_details()
		detail_name.text = "Ô trang bị %s (Đang trống)" % ("Vũ khí" if slot_type == "weapon" else "Phụ kiện")
		return

	var def := ItemDatabase.get_item(item_id)
	if def:
		_display_item_details(def, 1, true)

func _display_item_details(def: ItemDefinition, amount: int, is_equipped_slot: bool) -> void:
	if detail_icon:
		detail_icon.texture = def.icon
	if detail_name:
		detail_name.text = "%s (%d)" % [def.display_name, amount] if amount > 1 else def.display_name
	if detail_type:
		detail_type.text = "Loại: %s | Giá trị: %d G" % [def.get_type_name(), def.sell_value]
	if detail_desc:
		detail_desc.text = def.description

	var stat_text := ""
	if def.item_type == ItemDefinition.ItemType.WEAPON:
		stat_text = "+%d Sát thương đánh thường" % int(def.stat_modifiers.get("attack_bonus", 0))
	elif def.item_type == ItemDefinition.ItemType.ACCESSORY:
		stat_text = "+%d Max MP" % int(def.stat_modifiers.get("max_mana_bonus", 0))
	elif def.item_type == ItemDefinition.ItemType.CONSUMABLE:
		stat_text = "Hồi %d HP" % int(def.use_effect.get("heal_hp", 0))
	if detail_stats:
		detail_stats.text = stat_text

	# Cập nhật trạng thái các nút
	if is_equipped_slot:
		if use_btn: use_btn.disabled = true
		if equip_btn: equip_btn.disabled = true
		if unequip_btn: unequip_btn.disabled = false
		if drop_btn: drop_btn.disabled = true
	else:
		if use_btn:
			use_btn.disabled = (def.item_type != ItemDefinition.ItemType.CONSUMABLE)
		if equip_btn:
			equip_btn.disabled = not (def.item_type == ItemDefinition.ItemType.WEAPON or def.item_type == ItemDefinition.ItemType.ACCESSORY)
		if unequip_btn:
			unequip_btn.disabled = true
		if drop_btn:
			drop_btn.disabled = false

	if detail_name: detail_name.text = "Chọn một vật phẩm để xem chi tiết"
	if detail_type: detail_type.text = ""
	if detail_desc: detail_desc.text = ""
	if detail_stats: detail_stats.text = ""

		return

func _on_equip_pressed() -> void:
	if _selected_slot_idx < 0 or player == null:
		return
	var inv := player.inventory_component
	var eq := player.equipment_component
	if inv == null or eq == null:
		return

	var s := inv.get_slot(_selected_slot_idx)
	if s.item_id.is_empty():
		return

	var def := ItemDatabase.get_item(s.item_id)
	if def == null:
		return

	var item_to_equip: String = s.item_id
	var prev_item := eq.equip(item_to_equip)
	inv.remove_item_at(_selected_slot_idx, 1)

	if not prev_item.is_empty():
		inv.add_item(prev_item, 1)

	refresh()

func _on_unequip_pressed() -> void:
	if _selected_equip_slot.is_empty() or player == null:
		return
	var inv := player.inventory_component
	var eq := player.equipment_component
	if inv == null or eq == null:
		return

	var item_id := eq.get_equipped(_selected_equip_slot)
	if item_id.is_empty():
		return

	if not inv.can_add_item(item_id, 1):
		var ui = get_parent() as GameUI
		if ui:
			ui.show_notification("Túi đồ đã đầy! Không thể tháo trang bị.")
		return

	var unequipped := eq.unequip(_selected_equip_slot)
	if not unequipped.is_empty():
		inv.add_item(unequipped, 1)

	refresh()

func _on_drop_pressed() -> void:
	if _selected_slot_idx < 0 or player == null:
		return
	var inv := player.inventory_component
	if inv == null:
		return

	var s := inv.get_slot(_selected_slot_idx)
	if s.item_id.is_empty() or s.amount <= 0:
		return

	var drop_id: String = s.item_id
	inv.remove_item_at(_selected_slot_idx, 1)

	# Sinh ItemPickup ra thế giới
	var pickup_scene: PackedScene = preload("res://src/loot/item_pickup.tscn")
	var pickup := pickup_scene.instantiate() as ItemPickup
	pickup.item_id = drop_id
	pickup.amount = 1
	var drop_pos := player.global_position + Vector2(float(player.facing_direction) * 45.0, 0.0)
	pickup.global_position = drop_pos
	if player.get_parent():
		player.get_parent().add_child(pickup)

	refresh()

	var def := ItemDatabase.get_item(item_id)
	if def:
		if label:
			if amount > 1:

				label.text = def.display_name
		if icon_sprite and def.icon:
			icon_sprite.texture = def.icon

	# Hiệu ứng trôi nổi nhẹ nhàng (bobbing effect)
	_anim_time += delta * 3.5
	position.y = _base_y + sin(_anim_time) * 4.0

	if is_picked_up:
		return

	if body is Player:
		var inv := body.get_node_or_null("InventoryComponent") as InventoryComponent

		if inv.can_add_item(item_id, amount):
			is_picked_up = true
			var left := inv.add_item(item_id, amount)

			var item_name := def.display_name if def else item_id

		else:

	return null

## Bảng rơi vật phẩm (Loot Table) cho Enemy trong NOCTELYRA (M3)

# Mảng các quy tắc rơi: [{ "item_id": String, "chance": float, "min_amount": int, "max_amount": int }]
@export var drop_entries: Array[Dictionary] = [
	{
		"item_id": "mana_shard",
		"chance": 1.0, # 100% tỷ lệ rơi để kiểm thử nghiệm thu M3
		"min_amount": 1,
		"max_amount": 1
	}
]

func roll_drops() -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for entry in drop_entries:
		var chance: float = float(entry.get("chance", 1.0))
		if randf() <= chance:
			var min_amt: int = int(entry.get("min_amount", 1))
			var max_amt: int = int(entry.get("max_amount", 1))
			var amt := randi_range(min_amt, max_amt)
			if amt > 0:
				result.append({
					"item_id": str(entry.get("item_id", "mana_shard")),
					"amount": amt
				})
	return result

func spawn_loot(pos: Vector2, parent_node: Node) -> void:
	if parent_node == null:
		return
	var pickup_scene: PackedScene = preload("res://src/loot/item_pickup.tscn")
	var items := roll_drops()
	for i in range(items.size()):
		var item_info := items[i]
		var pickup := pickup_scene.instantiate() as ItemPickup
		pickup.item_id = item_info["item_id"]
		pickup.amount = item_info["amount"]
		pickup.is_dynamic = true
		pickup.pickup_id = "drop_" + str(Time.get_ticks_usec()) + "_" + str(randi() % 100000)
		pickup.global_position = pos + Vector2(float(i * 24 - (items.size() - 1) * 12), 0.0)
		parent_node.add_child.call_deferred(pickup)

	if current_quest == null:
		current_quest = QuestData.new()

func start_quest() -> bool:
	if current_quest == null or current_quest.current_state != QuestData.State.NOT_STARTED:
		return false
	current_quest.current_state = QuestData.State.ACTIVE
	current_quest.current_amount = 0
	_killed_enemy_ids.clear()
	emit_signal("quest_started", current_quest)
	emit_signal("notification_posted", "Đã nhận nhiệm vụ: %s" % current_quest.title)
	return true

func register_enemy_kill(enemy: Node) -> void:
	if current_quest == null or current_quest.current_state != QuestData.State.ACTIVE:
		return
	
	var kill_id: String = ""
	if enemy != null and "enemy_id" in enemy and not str(enemy.enemy_id).is_empty():
		kill_id = str(enemy.enemy_id)
	elif enemy != null:
		kill_id = "inst_" + str(enemy.get_instance_id())
	else:
		return

	if kill_id in _killed_enemy_ids:
		return # Đảm bảo mỗi quái chỉ tính 1 lần theo ID ổn định
	
	_killed_enemy_ids.append(kill_id)
	current_quest.current_amount += 1
	emit_signal("quest_progress_updated", current_quest)
	emit_signal("notification_posted", "Tiến độ: %d/%d sinh vật tha hóa" % [current_quest.current_amount, current_quest.target_amount])

	if current_quest.current_amount >= current_quest.target_amount:
		current_quest.current_state = QuestData.State.READY_TO_TURN_IN
		emit_signal("quest_ready_to_turn_in", current_quest)
		emit_signal("notification_posted", "Nhiệm vụ hoàn tất! Hãy về gặp Elyra để nhận thưởng.")

func complete_quest(player_health: HealthComponent = null) -> bool:
	if current_quest == null or current_quest.current_state != QuestData.State.READY_TO_TURN_IN:
		return false
	
	current_quest.current_state = QuestData.State.COMPLETED
	gold += current_quest.reward_gold
	emit_signal("gold_changed", gold)

	if player_health:
		player_health.current_health = player_health.max_health
		player_health.current_mana = player_health.max_mana
		player_health.emit_signal("health_changed", player_health.current_health, player_health.max_health)
		player_health.emit_signal("mana_changed", player_health.current_mana, player_health.max_mana)

	emit_signal("quest_completed", current_quest)
	emit_signal("notification_posted", "Hoàn thành: Nhận %d Vàng & hồi đầy HP/MP!" % current_quest.reward_gold)
	return true

class_name SaveManager

## Hệ thống Lưu & Tải Tiến trình (Save/Load System) cho NOCTELYRA (M3.1 FINAL)
## Hỗ trợ ghi tạm nguyên tử (Atomic write), backup an toàn, validate/clamping sâu,

const DEFAULT_SAVE_PATH: String = "user://noctelyra_slot1.json"
const CURRENT_VERSION: String = "1.0.0"

static var custom_save_path: String = ""
static var should_load_on_start: bool = false

	var test_json := JSON.new()
	var err := test_json.parse(text)
	if err != OK or not (test_json.data is Dictionary):

static func save_game(world: Node) -> bool:
	if world == null:
		return false

	var player := world.get_node_or_null("Player") as Player
	var quest_mgr := world.get_node_or_null("QuestManager") as QuestManager
	var ui := world.get_node_or_null("GameUI") as GameUI

	if player == null or quest_mgr == null:
		return false

	var p_health := player.get_node_or_null("HealthComponent") as HealthComponent
	var p_inv := player.get_node_or_null("InventoryComponent") as InventoryComponent
	var p_eq := player.get_node_or_null("EquipmentComponent") as EquipmentComponent

	var save_path := get_save_path()
	var tmp_path := get_tmp_path()
	var bak_path := get_bak_path()

	var save_data: Dictionary = {
		"save_version": CURRENT_VERSION,
		"timestamp": Time.get_unix_time_from_system(),
		"player": {
			"position_x": player.global_position.x,
			"position_y": player.global_position.y,
			"current_health": clamp(p_health.current_health, 0.0, p_health.max_health) if p_health else 100.0,
			"current_mana": clamp(p_health.current_mana, 0.0, p_health.max_mana) if p_health else 60.0
		},
		"gold": max(0, quest_mgr.gold),
		"inventory": p_inv.to_dict() if p_inv else [],
		"equipment": p_eq.to_dict() if p_eq else {},
		"quest": {
			"state": int(quest_mgr.current_quest.current_state) if quest_mgr.current_quest else 0,
			"current_amount": int(quest_mgr.current_quest.current_amount) if quest_mgr.current_quest else 0
		},

		"collected_pickup_ids": world.get("collected_pickup_ids") if "collected_pickup_ids" in world else [],
		"dropped_pickups": world.get_dropped_pickups_data() if world.has_method("get_dropped_pickups_data") else []
	}

	var json_str := JSON.stringify(save_data, "\t")

	# 1. Ghi an toàn vào file tạm .tmp
	var tmp_file := FileAccess.open(tmp_path, FileAccess.WRITE)
	if tmp_file == null:
		push_error("Không thể mở file tạm để ghi: " + tmp_path)
		return false

	tmp_file.store_string(json_str)
	tmp_file.flush()
	tmp_file.close()

	# 2. Xác minh dữ liệu trong file tạm trước khi chạm vào file cũ
	var verify_file := FileAccess.open(tmp_path, FileAccess.READ)
	if verify_file == null:
		DirAccess.remove_absolute(tmp_path)
		return false
	var verify_text := verify_file.get_as_text()
	verify_file.close()

	var verify_json := JSON.new()
	if verify_json.parse(verify_text) != OK or not (verify_json.data is Dictionary):
		push_error("Xác minh file tạm thất bại! Giữ nguyên bản lưu cũ.")
		DirAccess.remove_absolute(tmp_path)
		return false

	# 3. Thay thế an toàn không xóa mất file cũ (Bảo vệ dữ liệu M3.1)
	var has_old := FileAccess.file_exists(save_path)
	if has_old:
		if FileAccess.file_exists(bak_path):
			DirAccess.remove_absolute(bak_path)
		var bak_err := DirAccess.rename_absolute(save_path, bak_path)
		if bak_err != OK:
			# Không thể backup file cũ thì copy
			DirAccess.copy_absolute(save_path, bak_path)

	# Đổi tên .tmp -> file chính
	var replace_err := DirAccess.rename_absolute(tmp_path, save_path)
	if replace_err != OK:
		# Thử sao chép nếu rename thất bại
		var copy_err := DirAccess.copy_absolute(tmp_path, save_path)
		if copy_err != OK:

			if FileAccess.file_exists(bak_path):
				DirAccess.copy_absolute(bak_path, save_path)
			DirAccess.remove_absolute(tmp_path)
			return false
		DirAccess.remove_absolute(tmp_path)

	# Thành công, xóa file backup
	if FileAccess.file_exists(bak_path):
		DirAccess.remove_absolute(bak_path)

	if ui:
		ui.show_notification("Đã lưu tiến trình game thành công!")

	return true

static func load_game(world: Node) -> bool:
	if world == null or not has_save():
		return false

	var save_path := get_save_path()
	var file := FileAccess.open(save_path, FileAccess.READ)
	if file == null:
		return false
	var text := file.get_as_text()
	file.close()

	var test_json := JSON.new()
	var err := test_json.parse(text)
	if err != OK or not (test_json.data is Dictionary):
		push_warning("File save hỏng hoặc sai cú pháp JSON, bỏ qua để bảo vệ game.")
		return false

	var data: Dictionary = test_json.data
	var version: String = str(data.get("save_version", ""))
	if version.is_empty():
		push_warning("Thiếu phiên bản save_version.")
		return false

	var player := world.get_node_or_null("Player") as Player
	var quest_mgr := world.get_node_or_null("QuestManager") as QuestManager
	var ui := world.get_node_or_null("GameUI") as GameUI

	# 1. Khôi phục & Validate Player (Tọa độ & HP/MP)
	if player and data.has("player") and (data["player"] is Dictionary):
		var p_data: Dictionary = data["player"]
		var raw_x = p_data.get("position_x", player.global_position.x)
		var raw_y = p_data.get("position_y", player.global_position.y)

		var px: float = float(raw_x) if typeof(raw_x) in [TYPE_FLOAT, TYPE_INT] else player.global_position.x
		var py: float = float(raw_y) if typeof(raw_y) in [TYPE_FLOAT, TYPE_INT] else player.global_position.y
		if not is_finite(px): px = 180.0
		if not is_finite(py): py = 630.0

		# Giới hạn tọa độ trong thế giới an toàn (clamp)
		px = clamp(px, 50.0, 4300.0)
		py = clamp(py, 100.0, 700.0)
		player.global_position = Vector2(px, py)

		var p_health := player.get_node_or_null("HealthComponent") as HealthComponent
		if p_health:
			var raw_hp = p_data.get("current_health", p_health.max_health)
			var hp_val: float = float(raw_hp) if typeof(raw_hp) in [TYPE_FLOAT, TYPE_INT] else p_health.max_health
			p_health.current_health = clamp(hp_val, 0.0, p_health.max_health)

			var raw_mp = p_data.get("current_mana", p_health.max_mana)
			var mp_val: float = float(raw_mp) if typeof(raw_mp) in [TYPE_FLOAT, TYPE_INT] else p_health.max_mana
			p_health.current_mana = clamp(mp_val, 0.0, p_health.max_mana)

			p_health.emit_signal("health_changed", p_health.current_health, p_health.max_health)
			p_health.emit_signal("mana_changed", p_health.current_mana, p_health.max_mana)

	# 2. Khôi phục & Validate Inventory & Equipment
	if player:
		var p_inv := player.get_node_or_null("InventoryComponent") as InventoryComponent
		if p_inv and data.has("inventory") and (data["inventory"] is Array):
			p_inv.load_from_dict(data["inventory"])

		var p_eq := player.get_node_or_null("EquipmentComponent") as EquipmentComponent
		if p_eq and data.has("equipment") and (data["equipment"] is Dictionary):
			p_eq.load_from_dict(data["equipment"])

	# 3. Khôi phục & Validate Vàng & Nhiệm vụ
	if quest_mgr:
		var raw_gold = data.get("gold", 0)
		var val_gold: int = int(raw_gold) if typeof(raw_gold) in [TYPE_INT, TYPE_FLOAT] else 0
		quest_mgr.gold = clamp(val_gold, 0, 999999)
		quest_mgr.emit_signal("gold_changed", quest_mgr.gold)

		if data.has("quest") and (data["quest"] is Dictionary):
			var q_data: Dictionary = data["quest"]
			var raw_state = q_data.get("state", 0)
			var q_state: int = int(raw_state) if typeof(raw_state) in [TYPE_INT, TYPE_FLOAT] else 0
			q_state = clamp(q_state, 0, 3)

			var raw_amt = q_data.get("current_amount", 0)
			var q_amt: int = int(raw_amt) if typeof(raw_amt) in [TYPE_INT, TYPE_FLOAT] else 0

			if quest_mgr.current_quest:
				quest_mgr.current_quest.current_state = q_state as QuestData.State
				quest_mgr.current_quest.current_amount = clamp(q_amt, 0, quest_mgr.current_quest.target_amount)
				if q_state == QuestData.State.COMPLETED:
					quest_mgr.current_quest.current_amount = quest_mgr.current_quest.target_amount

				if ui:
					ui.update_quest_info(quest_mgr.current_quest)
					ui.update_gold(quest_mgr.gold)

	# 4. Khôi phục trạng thái vật phẩm thế giới và quái vật (M3.1)
	var raw_collected = data.get("collected_pickup_ids", [])
	var collected_ids: Array = raw_collected if raw_collected is Array else []

	var raw_dropped = data.get("dropped_pickups", [])
	var dropped_pickups: Array = raw_dropped if raw_dropped is Array else []

	var raw_defeated = data.get("defeated_enemy_ids", [])
	var defeated_ids: Array = raw_defeated if raw_defeated is Array else []

	if world.has_method("restore_world_state"):
		world.restore_world_state(collected_ids, dropped_pickups, defeated_ids)

	# Đồng bộ danh sách quái đã chết vào QuestManager để chống đếm trùng
	if quest_mgr:
		for did in defeated_ids:
			var sid := str(did)
			if not sid in quest_mgr._killed_enemy_ids:
				quest_mgr._killed_enemy_ids.append(sid)

	if ui:
		ui.show_notification("Đã tải bản lưu thành công!")

	return true

func _unhandled_input(event: InputEvent) -> void:
	# Mở/đóng Túi đồ bằng phím I
	if event.is_action_pressed("inventory"):
		if not is_dialogue_open():
			toggle_inventory()
			get_viewport().set_input_as_handled()
			return

	# Chặn các phím hành động khi đang mở hội thoại, trong thời gian hồi, hoặc đang mở túi đồ
	if is_dialogue_open() or is_dialogue_cooling_down() or is_inventory_open():
		if event.is_action_pressed("attack") or event.is_action_pressed("jump") or event.is_action_pressed("cast_spell") or event.is_action_pressed("dash"):
			get_viewport().set_input_as_handled()
			return

	# Dự phòng nếu không có InteractionCoordinator trong scene
	if is_dialogue_open() and event.is_action_pressed("interact"):
		if get_tree() and get_tree().get_nodes_in_group("interaction_coordinator").size() == 0:
			advance_dialogue()

	if quest_mgr:

const GAME_SCENE_PATH: String = "res://src/world/noctis_forest.tscn"

@onready var continue_btn: Button = $Panel/VBox/ContinueBtn
@onready var new_game_btn: Button = $Panel/VBox/NewGameBtn
@onready var quit_btn: Button = $Panel/VBox/QuitBtn

@onready var confirm_dialog: Control = $ConfirmDialog
@onready var confirm_yes_btn: Button = $ConfirmDialog/Panel/HBox/YesBtn
@onready var confirm_no_btn: Button = $ConfirmDialog/Panel/HBox/NoBtn

func _ready() -> void:
	if confirm_dialog:
		confirm_dialog.visible = false

	_update_buttons()

	if continue_btn:
		continue_btn.pressed.connect(_on_continue_pressed)
	if new_game_btn:
		new_game_btn.pressed.connect(_on_new_game_pressed)
	if quit_btn:
		quit_btn.pressed.connect(_on_quit_pressed)

	if confirm_yes_btn:
		confirm_yes_btn.pressed.connect(_on_confirm_new_game)
	if confirm_no_btn:
		confirm_no_btn.pressed.connect(func(): confirm_dialog.visible = false)

func _update_buttons() -> void:
	var has_valid_save := SaveManager.has_save()
	if continue_btn:
		continue_btn.disabled = not has_valid_save

func _on_continue_pressed() -> void:
	if SaveManager.has_save():
		SaveManager.should_load_on_start = true
		get_tree().change_scene_to_file(GAME_SCENE_PATH)

func _on_new_game_pressed() -> void:
	if SaveManager.has_save():
		if confirm_dialog:
			confirm_dialog.visible = true
	else:
		_start_new_game()

func _on_confirm_new_game() -> void:
	SaveManager.delete_save()
	_start_new_game()

func _start_new_game() -> void:
	SaveManager.should_load_on_start = false
	get_tree().change_scene_to_file(GAME_SCENE_PATH)

func _on_quit_pressed() -> void:
	get_tree().quit()

class_name PauseController
extends Node

@export var pause_menu: Control

func _ready() -> void:
	# PauseController luôn luôn xử lý input bất kể game đang pause hay không
	process_mode = Node.PROCESS_MODE_ALWAYS
	if pause_menu:
		pause_menu.visible = false

func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed("pause"):
		# Nếu túi đồ đang mở thì Esc đóng túi đồ trước thay vì mở PauseMenu
		var cur_scene = get_tree().current_scene if get_tree() else null
		if cur_scene:
			var ui := cur_scene.get_node_or_null("GameUI") as GameUI
			if ui and ui.is_inventory_open():
				ui.close_inventory()
				get_viewport().set_input_as_handled()
				return

		toggle_pause()
		get_viewport().set_input_as_handled()

func toggle_pause() -> void:
	var tree := get_tree()
	if tree == null:
		return
	if tree.paused:
		resume()
	else:
		pause()

func pause() -> void:
	var tree := get_tree()
	if tree == null:
		return
	tree.paused = true
	if pause_menu:
		pause_menu.visible = true

func resume() -> void:
	var tree := get_tree()
	if tree == null:
		return
	tree.paused = false
	if pause_menu:
		pause_menu.visible = false

class_name NoctisForest
extends Node2D

var player: Player
var game_ui: GameUI
var quest_manager: QuestManager
var pause_controller: PauseController
var npc_elyra: NPC
var interaction_coordinator: InteractionCoordinator
var defeated_enemy_ids: Array = []
var collected_pickup_ids: Array = []

func _enter_tree() -> void:
	init_references()

func _ready() -> void:
	process_mode = Node.PROCESS_MODE_PAUSABLE
	init_references()
	if SaveManager.should_load_on_start:
		SaveManager.should_load_on_start = false
		SaveManager.load_game(self)

func init_references() -> void:
	if player == null:
		player = get_node_or_null("Player") as Player
	if game_ui == null:
		game_ui = get_node_or_null("GameUI") as GameUI
	if quest_manager == null:
		quest_manager = get_node_or_null("QuestManager") as QuestManager
	if pause_controller == null:
		pause_controller = get_node_or_null("PauseController") as PauseController
	if npc_elyra == null:
		npc_elyra = get_node_or_null("NPC_Elyra") as NPC
	if interaction_coordinator == null:
		interaction_coordinator = get_node_or_null("InteractionCoordinator") as InteractionCoordinator

	# Kết nối Player
	if player and not player.state_changed.is_connected(_on_player_state_changed):
		player.add_to_group("players")
		player.state_changed.connect(_on_player_state_changed)
		_on_player_state_changed("IDLE")

		var p_health: HealthComponent = player.get_node_or_null("HealthComponent")
		if p_health and not p_health.health_changed.is_connected(_on_player_health_changed):
			p_health.health_changed.connect(_on_player_health_changed)
			p_health.mana_changed.connect(_on_player_mana_changed)
			if game_ui:
				game_ui.update_player_hp(p_health.current_health, p_health.max_health)
				game_ui.update_player_mp(p_health.current_mana, p_health.max_mana)

	# Kết nối GameUI
	if game_ui and not game_ui.retry_requested.is_connected(_on_retry_requested):
		game_ui.retry_requested.connect(_on_retry_requested)
		if player and game_ui.inventory_ui:
			game_ui.inventory_ui.setup_player(player)
		if quest_manager and quest_manager.current_quest:
			game_ui.update_quest_info(quest_manager.current_quest)
			game_ui.update_gold(quest_manager.gold)

	# Kết nối QuestManager
	if quest_manager and not quest_manager.quest_started.is_connected(_on_quest_updated):
		quest_manager.quest_started.connect(_on_quest_updated)
		quest_manager.quest_progress_updated.connect(_on_quest_updated)
		quest_manager.quest_ready_to_turn_in.connect(_on_quest_updated)
		quest_manager.quest_completed.connect(_on_quest_updated)
		quest_manager.gold_changed.connect(_on_gold_changed)
		quest_manager.notification_posted.connect(_on_notification_posted)

	# Kết nối NPC Elyra
	if npc_elyra and not npc_elyra.dialogue_requested.is_connected(_on_npc_dialogue_requested):
		npc_elyra.dialogue_requested.connect(_on_npc_dialogue_requested)

	# Kết nối các quái vật trên map với QuestManager và ghi nhận ID quái đã diệt
	var enemies = get_tree().get_nodes_in_group("enemies")
	for e in enemies:
		if e is Enemy:
			if quest_manager and not e.enemy_defeated.is_connected(quest_manager.register_enemy_kill):
				e.enemy_defeated.connect(quest_manager.register_enemy_kill)
			if not e.enemy_defeated.is_connected(_on_enemy_defeated):
				e.enemy_defeated.connect(_on_enemy_defeated)

func _on_enemy_defeated(enemy_node: Node) -> void:
	if enemy_node is Enemy and not enemy_node.enemy_id.is_empty():
		if not enemy_node.enemy_id in defeated_enemy_ids:
			defeated_enemy_ids.append(enemy_node.enemy_id)

func register_collected_pickup(p_id: String) -> void:
	if not p_id.is_empty() and not p_id in collected_pickup_ids:
		collected_pickup_ids.append(p_id)

func get_dropped_pickups_data() -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	var dyn_pickups = get_tree().get_nodes_in_group("dynamic_pickups")
	for p in dyn_pickups:
		if p is ItemPickup and is_instance_valid(p) and not p.is_picked_up and not p.is_queued_for_deletion():
			result.append({
				"pickup_id": p.pickup_id,
				"item_id": p.item_id,
				"amount": p.amount,
				"x": p.global_position.x,
				"y": p.global_position.y
			})
	return result

func restore_world_state(p_collected_ids: Array, p_dropped_pickups: Array, p_defeated_ids: Array) -> void:
	collected_pickup_ids.clear()
	for id in p_collected_ids:
		collected_pickup_ids.append(str(id))

	defeated_enemy_ids.clear()
	for id in p_defeated_ids:
		defeated_enemy_ids.append(str(id))

	# 1. Dọn các vật phẩm đặt sẵn đã từng nhặt
	var preplaced = get_tree().get_nodes_in_group("preplaced_pickups")
	for p in preplaced:
		if p is ItemPickup and not p.pickup_id.is_empty():
			if p.pickup_id in collected_pickup_ids:
				p.queue_free()

	# 2. Xóa các dynamic pickup cũ trước khi phục hồi danh sách lưu (tránh nhân bản)
	var existing_dyn = get_tree().get_nodes_in_group("dynamic_pickups")
	for p in existing_dyn:
		if is_instance_valid(p):
			p.queue_free()

	# 3. Phục hồi các vật phẩm quái rơi chưa nhặt
	var pickup_scene: PackedScene = preload("res://src/loot/item_pickup.tscn")
	for dp in p_dropped_pickups:
		if dp is Dictionary:
			var p := pickup_scene.instantiate() as ItemPickup
			p.pickup_id = str(dp.get("pickup_id", ""))
			p.item_id = str(dp.get("item_id", "mana_shard"))
			p.amount = max(1, int(dp.get("amount", 1)))
			p.is_dynamic = true
			var px: float = float(dp.get("x", 0.0))
			var py: float = float(dp.get("y", 0.0))
			p.global_position = Vector2(px, py)
			add_child(p)

	# 4. Xóa các quái đã chết
	var enemies = get_tree().get_nodes_in_group("enemies")
	for e in enemies:
		if e is Enemy and not e.enemy_id.is_empty():
			if e.enemy_id in defeated_enemy_ids:
				e.queue_free()

func _on_player_state_changed(st_name: String) -> void:
	if game_ui:
		game_ui.update_player_state(st_name)

func _on_player_health_changed(cur: float, max_v: float) -> void:
	if game_ui:
		game_ui.update_player_hp(cur, max_v)

func _on_player_mana_changed(cur: float, max_v: float) -> void:
	if game_ui:
		game_ui.update_player_mp(cur, max_v)

func _on_quest_updated(quest: QuestData) -> void:
	if game_ui:
		game_ui.update_quest_info(quest)

func _on_gold_changed(new_gold: int) -> void:
	if game_ui:
		game_ui.update_gold(new_gold)

func _on_notification_posted(msg: String) -> void:
	if game_ui:
		game_ui.show_notification(msg)

func _on_npc_dialogue_requested(npc: NPC, lines: Array[String], action: String) -> void:
	if game_ui:
		game_ui.show_dialogue(npc.npc_name, lines, action)

func _on_retry_requested() -> void:
	if player == null:
		player = get_node_or_null("Player") as Player
	if player:
		player.respawn()

func _on_resume_button_pressed() -> void:
	if pause_controller:
		pause_controller.resume()

func _on_quit_button_pressed() -> void:
	get_tree().quit()

extends Node2D

@onready var player: Player = $Player
@onready var state_label: Label = $CanvasLayer/HUD/StateLabel
@onready var hp_bar: ProgressBar = $CanvasLayer/HUD/VBoxBars/HPBar
@onready var mp_bar: ProgressBar = $CanvasLayer/HUD/VBoxBars/MPBar
@onready var pause_controller: PauseController = $PauseController

func _ready() -> void:
	# Gameplay level đặt PROCESS_MODE_PAUSABLE
	process_mode = Node.PROCESS_MODE_PAUSABLE

	if player:
		# Thêm Player vào group "players" để Enemy dễ nhận diện
		player.add_to_group("players")
		player.state_changed.connect(_on_player_state_changed)
		_on_player_state_changed("IDLE")

		var health_comp: HealthComponent = player.get_node_or_null("HealthComponent")
		if health_comp:
			health_comp.health_changed.connect(_on_player_health_changed)
			health_comp.mana_changed.connect(_on_player_mana_changed)
			_on_player_health_changed(health_comp.current_health, health_comp.max_health)
			_on_player_mana_changed(health_comp.current_mana, health_comp.max_mana)

func _on_player_state_changed(new_state: String) -> void:
	if state_label:
		state_label.text = "State: %s" % new_state

func _on_player_health_changed(cur: float, max_v: float) -> void:
	if hp_bar:
		hp_bar.max_value = max_v
		hp_bar.value = cur

func _on_player_mana_changed(cur: float, max_v: float) -> void:
	if mp_bar:
		mp_bar.max_value = max_v
		mp_bar.value = cur

func _on_resume_button_pressed() -> void:
	var ctrl = pause_controller if pause_controller else get_node_or_null("PauseController")
	if ctrl:
		ctrl.resume()
	else:
		get_tree().paused = false
		var menu = get_node_or_null("CanvasLayer/PauseMenu")
		if menu:
			menu.visible = false

func _on_quit_button_pressed() -> void:
	get_tree().quit()
